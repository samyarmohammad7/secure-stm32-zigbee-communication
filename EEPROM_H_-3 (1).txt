#ifndef INC_EEPROM_H_
#define INC_EEPROM_H_

#include "main.h"
#include <stdint.h>

#define EEPROM_I2C_ADDRESS       (0x50 << 1)  // 24LC64, A0/A1/A2 tied low
#define EEPROM_TOTAL_BYTES       8192U        // 64 kbit / 8 kB
#define EEPROM_PAGE_SIZE         32U

#define EEPROM_THREAT_BASE_ADDR  32U          // Threat table starts after page 0
#define EEPROM_THREAT_RECORD_LEN 4U           // [sector, flags, severity, seq]

HAL_StatusTypeDef EEPROM_ReadBytes(uint16_t mem_addr,
                                   uint8_t *data,
                                   uint16_t length);

HAL_StatusTypeDef EEPROM_WriteBytes(uint16_t mem_addr,
                                    const uint8_t *data,
                                    uint16_t length);

HAL_StatusTypeDef EEPROM_StoreThreat(uint8_t sector,
                                     uint8_t flags,
                                     uint8_t severity,
                                     uint8_t seq);

HAL_StatusTypeDef EEPROM_ReadThreat(uint8_t sector,
                                    uint8_t *record_out);

#endif /* INC_EEPROM_H_ */
